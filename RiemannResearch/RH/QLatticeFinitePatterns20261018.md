(* Reproducible Wolfram Language checks, 2026-10-18. Each Module self-contained.
   Numeric Gram experiments are labeled numerical. Final certificate checks use exact rational arithmetic only. *)

(* Subset convolution exp/log: m=2,3,4 all true. *)
Module[{conv,subs,unit,h,terms,ex,inc,lg,m,full,product,generic,gexp,glog,result},conv[f_,g_,mm_]:=Table[Total[Table[If[BitAnd[t,s]==t,f[[t+1]] g[[BitXor[s,t]+1]],0],{t,0,2^mm-1}]],{s,0,2^mm-1}];result=Table[unit=UnitVector[2^m,1];h=Table[If[DigitCount[s,2,1]==1,1+IntegerExponent[s,2],0],{s,0,2^m-1}];terms=NestList[conv[#,h,m]&,unit,m];ex=Sum[terms[[k+1]]/k!,{k,0,m}];product=Table[Times@@Pick[Range[m],IntegerDigits[s,2,m]//Reverse,1],{s,0,2^m-1}];inc=product-unit;terms=NestList[conv[#,inc,m]&,unit,m];lg=Sum[(-1)^(k+1) terms[[k+1]]/k,{k,1,m}];generic=Table[s/(s+1),{s,0,2^m-1}];terms=NestList[conv[#,generic,m]&,unit,m];gexp=Sum[terms[[k+1]]/k!,{k,0,m}];terms=NestList[conv[#,gexp-unit,m]&,unit,m];glog=Sum[(-1)^(k+1) terms[[k+1]]/k,{k,1,m}];{m,ex==product,lg==h,glog==generic,Last[ex],Last[lg]},{m,{2,3,4}}];result]

Module[{m,zeta,cover,identity,logz,mobius,results,a,b,minor},results=Table[zeta=Table[Boole[BitAnd[t,s]==t],{s,0,2^m-1},{t,0,2^m-1}];cover=Table[Boole[BitAnd[t,s]==t && DigitCount[s,2,1]-DigitCount[t,2,1]==1],{s,0,2^m-1},{t,0,2^m-1}];identity=IdentityMatrix[2^m];logz=Sum[(-1)^(k+1)MatrixPower[zeta-identity,k]/k,{k,1,m}];mobius=Table[If[BitAnd[t,s]==t,(-1)^(DigitCount[s,2,1]-DigitCount[t,2,1]),0],{s,0,2^m-1},{t,0,2^m-1}];{m,logz==cover,MatrixExp[cover]==zeta,MatrixExp[-cover]==mobius,MatrixPower[cover,m+1]==0 identity,Det[zeta],FullSimplify[Max[SingularValueList[N[zeta]]]]},{m,{2,3,4}}];minor=Det[Table[Append[Table[Mod[n,d]/d,{d,{2,3,6}}],Mod[n,2]Mod[n,3]/6],{n,1,4}]];{results,"NonlinearMinor"->minor,"OnePrimeSingularSquares"->Eigenvalues[{{1,1},{0,1}}.{{1,0},{1,1}}]}]

(* Full inverse-square Q regularization: numerical search, 50-digit working precision. *)
Module[{a=1/Zeta[2],primes,ds,period,atom,weights,gram,hessian,b,coef,alpha,energy,residual,penalty,results},results=Table[primes=Prime[Range[m]];period=Times@@primes;ds=Rest[Divisors[period]];atom=Table[Mod[j,d]/d,{j,1,period},{d,ds}];weights=N[Table[PolyGamma[1,j/period]/(Zeta[2]period^2),{j,1,period}],50];gram=Transpose[atom].DiagonalMatrix[weights].atom;hessian=Table[1/(d e)-1/(d e^2)-1/(e d^2)+1/LCM[d,e]^2,{d,ds},{e,ds}];b=a/ds;Table[coef=LinearSolve[N[hessian+penalty gram,50],N[b,50]];alpha=N[a,50]-N[b,50].coef;energy=coef.gram.coef;residual=alpha-penalty energy;{m,period,Length[ds],penalty,N[alpha/penalty,12],N[alpha,12],N[energy,12],N[residual,12]},{penalty,{1/10,1/100,1/1000,1/10000}}],{m,1,4}];Flatten[results,1]]
