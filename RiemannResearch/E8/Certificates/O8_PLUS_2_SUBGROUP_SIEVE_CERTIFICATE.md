# O8+(2) subgroup-sieve certificate

This file restores the script and raw output behind the exhaustive subgroup
sieve and perfect-core compression used in the Serre II E8 development.

## Trust boundary

Lean proves the general orbit-partition gcd theorem, the stable-derived-series
perfect-core theorems, and consistency properties of the embedded finite data.
The numerical identification of those data with TomLib's table of marks is a
rerunnable external certificate: it is not presented as a theorem obtained by
constructing the 174,182,400-element group inside Lean.

The eight TomLib column positions are a source-matched input. The script checks
their table-theoretic properties and exhausts every TomLib class relative to
those columns, but it does not construct all eight E8 root-subsystem actions
and derive their TomLib positions internally. Totaro supplies the
maximal-overgroup classification; the geometric column matching remains an
explicit external identification.

## Why the sieve is exhaustive

Let $Q = \Omega_8^+(2)$. Totaro's maximal-overgroup classification identifies
the subsystem stabilizers relevant to a proper root subsystem. Seven arise
from the displayed subsystem types and the eighth is the $(A_1)^8$ stabilizer.
The classification is naturally stated up to $O_8^+(2)$-conjugacy, but this
does not create missing $Q$-classes: each subsystem stabilizer contains an odd
root reflection $r$. If an odd element $g$ gives a conjugate, then $gr \in Q$
and $(gr)M(gr)^{-1} = gMg^{-1}$.

Once the eight geometric-to-TomLib column identifications are supplied, a
nonzero mark for a table-of-marks class $H$ in the coset action $Q/M$ is
equivalent to $H$ being subconjugate to $M$. Consequently, testing zero
marks in the eight correlated stabilizer columns tests every $Q$-conjugacy
class. Triality gives three correlated column choices; the script checks all
three. Each first-stage sieve has 313 survivors and its corresponding
$(A_1)^8$ column removes exactly ten, leaving 303.

The derived-subgroup map stored by TomLib is then iterated to stability. Of the
303 survivors, 150 reach the trivial class and 153 reach 31 nontrivial perfect
classes, with 14 distinct orders. Lean separately proves that any stable term
of the derived series is perfect, has solvable quotient, and is contained in
every normal subgroup with solvable quotient.

## GAP script

```gap
#############################################################################
##
##  Reproducible TomLib certificate for the O8+(2) subgroup sieve.
##
##  The three configurations below are permuted by triality.  Each consists
##  of the seven subsystem-stabilizer classes used in the original sieve,
##  followed by the matching A1^8 stabilizer class.
##
#############################################################################

if LoadPackage("tomlib") <> true then
    Error("TomLib failed to load");
fi;

tom := TableOfMarks("O8+(2)");
subs := SubsTom(tom);
orders := OrdersTom(tom);
derived := DerivedSubgroupsTomUnique(tom);
n := Length(subs);

Assert(0, Identifier(tom) = "O8+(2)");
Assert(0, n = 11171);
Assert(0, Length(NrSubsTom(tom)) = n);
Assert(0, Length(orders) = n);
Assert(0, Length(NormalizersTom(tom)) = n);
Assert(0, Length(derived) = n);
Assert(0, orders[n] = 174182400);

maximal := MaximalSubgroupsTom(tom)[1];
maximalNames := Maxes(CharacterTable("O8+(2)"));
fusionSourceTopClasses := NotifiedFusionsToLibTom("O8+(2)");
expectedMaximal := [
    11170, 11169, 11168,
    11167, 11166, 11165,
    11164, 11163, 11162,
    11158, 11157, 11156,
    11155, 11077,
    11076, 11075, 11074
];
Assert(0, maximal = expectedMaximal);
Assert(0, Length(maximalNames) = Length(maximal));

configurations := [
    rec(
        seven := [11170, 11165, 11164, 11156, 11074, 11155, 11077],
        a1e8 := 11152
    ),
    rec(
        seven := [11169, 11166, 11163, 11158, 11076, 11155, 11077],
        a1e8 := 11153
    ),
    rec(
        seven := [11168, 11167, 11162, 11157, 11075, 11155, 11077],
        a1e8 := 11154
    )
];

PerfectCoreClass := function(class)
    local next;
    while derived[class] <> class do
        next := derived[class];
        Assert(0, orders[next] < orders[class]);
        class := next;
    od;
    return class;
end;

coreIdentifiersByOrder := [
    [60,        "A5"],
    [360,       "A6"],
    [960,       "2^4:A5"],
    [1920,      "2^5:A5"],
    [2520,      "A7"],
    [3600,      "A5xA5"],
    [11520,     "2^5:A6"],
    [20160,     "A8"],
    [25920,     "U4(2)"],
    [161280,    "2^6:A7"],
    [181440,    "A9"],
    [1290240,   "2^6:A8"],
    [1451520,   "S6(2)"],
    [174182400, "O8+(2)"]
];

CoreIdentifier := function(order)
    local entry;
    entry := First(coreIdentifiersByOrder, pair -> pair[1] = order);
    Assert(0, entry <> fail);
    return entry[2];
end;

Print("GAP_VERSION=", GAPInfo.Version, "\n");
Print("TOMLIB_VERSION=", GAPInfo.PackagesLoaded.tomlib[2], "\n");
Print("TOMLIB_GIT_COMMIT=c460dc4f6bbfbf805b0f0f2dd1d0841bbd4e9c2c\n");
Print("TMO8P2_SHA256=eaa82cedb5b805047de0f859f7b17e3dc6c09f7878d4d97e604fa69054c905ed\n");
Print("GAP_TARBALL_SHA256=aaa296b32a5d7bf25fd80f241d23ec1f58b74e991ae730fafe40e54eb3af6e7e\n");
Print("TABLE=", Identifier(tom), "\n");
Print("GROUP_ORDER=", orders[n], "\n");
Print("SUBGROUP_CLASSES=", n, "\n");
Print("MAXIMAL_CLASS_POSITIONS=", maximal, "\n");
Print("MAXIMAL_CHARACTER_TABLE_NAMES=", maximalNames, "\n");
Print("FUSION_SOURCE_TOP_CLASSES=", fusionSourceTopClasses, "\n");

RunConfiguration := function(cfgno, cfg)
    local forbiddenSeven, class, survivorsSeven, forbiddenEight,
          survivors, cores, solvable, nonsolvable, perfectCoreClasses,
          perfectCoreOrders, removedByA1E8, perfectCoreData;

    forbiddenSeven := [];
    for class in cfg.seven do
        UniteSet(forbiddenSeven, subs[class]);
    od;
    survivorsSeven := Difference([1 .. n], forbiddenSeven);

    forbiddenEight := ShallowCopy(forbiddenSeven);
    UniteSet(forbiddenEight, subs[cfg.a1e8]);
    survivors := Difference([1 .. n], forbiddenEight);
    removedByA1E8 := Difference(survivorsSeven, survivors);

    cores := List(survivors, PerfectCoreClass);
    solvable := Filtered([1 .. Length(survivors)], i -> cores[i] = 1);
    nonsolvable := Filtered([1 .. Length(survivors)], i -> cores[i] <> 1);
    perfectCoreClasses := Set(cores{nonsolvable});
    perfectCoreOrders := Set(orders{perfectCoreClasses});
    perfectCoreData := List(perfectCoreClasses, class -> [
        class,
        orders[class],
        CoreIdentifier(orders[class]),
        Number(cores, core -> core = class)
    ]);

    Assert(0, ForAll(cfg.seven, class -> class in maximal));
    Assert(0, orders[cfg.a1e8] = 86016);
    Assert(0, derived[cfg.a1e8] = cfg.a1e8);
    Assert(0, Length(survivorsSeven) = 313);
    Assert(0, Length(survivors) = 303);
    Assert(0, Length(removedByA1E8) = 10);
    Assert(0, Length(solvable) = 150);
    Assert(0, Length(nonsolvable) = 153);
    Assert(0, ForAll(solvable,
        i -> ForAll(PrimeDivisors(orders[survivors[i]]),
                    p -> p in [2, 3, 5])));
    Assert(0, Length(perfectCoreClasses) = 31);
    Assert(0, ForAll(perfectCoreClasses,
                     class -> derived[class] = class));
    Assert(0, Length(perfectCoreOrders) = 14);

    Print("CONFIGURATION_", cfgno, "_SEVEN=", cfg.seven, "\n");
    Print("CONFIGURATION_", cfgno, "_A1E8=", cfg.a1e8, "\n");
    Print("CONFIGURATION_", cfgno, "_AFTER_SEVEN=",
          Length(survivorsSeven), "\n");
    Print("CONFIGURATION_", cfgno, "_AFTER_A1E8=",
          Length(survivors), "\n");
    Print("CONFIGURATION_", cfgno, "_REMOVED_BY_A1E8=",
          removedByA1E8, "\n");
    Print("CONFIGURATION_", cfgno, "_SURVIVOR_CLASS_POSITIONS=",
          survivors, "\n");
    Print("CONFIGURATION_", cfgno, "_SOLVABLE_235=",
          Length(solvable), "\n");
    Print("CONFIGURATION_", cfgno, "_NONSOLVABLE=",
          Length(nonsolvable), "\n");
    Print("CONFIGURATION_", cfgno, "_PERFECT_CORE_CLASSES=",
          perfectCoreClasses, "\n");
    Print("CONFIGURATION_", cfgno, "_PERFECT_CORE_ORDERS=",
          perfectCoreOrders, "\n");
    Print("CONFIGURATION_", cfgno, "_PERFECT_CORE_DATA=",
          perfectCoreData, "\n");
end;

for cfgno in [1 .. Length(configurations)] do
    RunConfiguration(cfgno, configurations[cfgno]);
od;

Print("CERTIFICATE_OK=true\n");
QUIT;
```
